const sleep = (n:number) => new Promise(r=>setTimeout(r,n));
function env(s:any, extra:any={}) { const e={...process.env}; for (const k of Object.keys(e)) if (/^(LIMEN_|HERDR_|PI_)/.test(k)) delete e[k]; return {...e, PATH:`${s.fakeBin}:${process.env.PATH}`, LIMEN_PI:'pi', LIMEN_HOME:dirname(s.root), LIMEN_HERDR:'0', LIMEN_HUNK:'0', ...extra}; }
function asyncLimen(s:any,args:string[],extra:any={}) { const p=spawn(process.execPath,[LIMEN,...args],{cwd:s.root,env:env(s,extra),stdio:['ignore','pipe','pipe']}); let stdout='',stderr=''; p.stdout.on('data',b=>stdout+=b); p.stderr.on('data',b=>stderr+=b); return new Promise<any>(r=>p.on('close',status=>r({status,stdout,stderr}))); }
